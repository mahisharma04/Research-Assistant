### `agents.py`

```python
"""
Research Assistant for College Students: 3-agent system built with LangGraph.

Agents:
  1. Researcher -> plans search queries, calls the web search tool, writes cited notes
  2. Writer     -> turns the notes into a structured summary (uses ONLY the notes)
  3. Reviewer   -> checks every claim against the notes; approves or sends back for revision

Flow:  Researcher -> Writer -> Reviewer
       Reviewer says REVISE (max 2 times) -> back to Writer
       Reviewer says APPROVED             -> END
"""
import json
import os
import re
from typing import List, TypedDict

from dotenv import load_dotenv
from ddgs import DDGS
from langchain_google_genai import ChatGoogleGenerativeAI
from langgraph.graph import END, StateGraph

load_dotenv()

MAX_REVISIONS = 2
MODEL = os.getenv("GEMINI_MODEL", "gemini-2.5-flash")  # reads GOOGLE_API_KEY from env


# ---------------------------------------------------------------- shared state
class ResearchState(TypedDict, total=False):
    topic: str
    queries: List[str]
    sources: List[dict]      # [{id, title, url, snippet}]
    notes: str
    draft: str
    verdict: str             # "APPROVED" or "REVISE"
    issues: List[str]
    revisions: int
    log: List[str]


# ---------------------------------------------------------------------- helpers
def _llm(temperature: float = 0.3) -> ChatGoogleGenerativeAI:
    return ChatGoogleGenerativeAI(model=MODEL, temperature=temperature)


def _ask(system: str, user: str, temperature: float = 0.3) -> str:
    resp = _llm(temperature).invoke([("system", system), ("human", user)])
    content = resp.content
    if isinstance(content, list):  # some versions return a list of parts
        content = "".join(p.get("text", "") if isinstance(p, dict) else str(p) for p in content)
    return content.strip()


# ------------------------------------------------------------------------ tool
def web_search(query: str, max_results: int = 4) -> List[dict]:
    """Tool used by the Researcher agent. Returns title, url, snippet."""
    try:
        results = DDGS().text(query, max_results=max_results)
    except Exception:
        return []
    return [
        {"title": r.get("title", ""), "url": r.get("href", ""), "snippet": r.get("body", "")}
        for r in results
    ]


# ----------------------------------------------------------------------- agents
def researcher(state: ResearchState) -> dict:
    topic = state["topic"]

    raw = _ask(
        "You plan web searches for a college student's research task. "
        "Return exactly 3 different, specific search queries, one per line, no numbering.",
        f"Topic: {topic}",
    )
    queries = [q.strip("-*0123456789. ").strip() for q in raw.splitlines() if q.strip()][:3]

    sources, seen = [], set()
    for q in queries:
        for r in web_search(q):
            if r["url"] and r["url"] not in seen:
                seen.add(r["url"])
                sources.append({"id": len(sources) + 1, **r})

    if not sources:
        return {
            "queries": queries, "sources": [], "notes": "",
            "log": state.get("log", []) + ["Researcher: no search results found."],
        }

    results_text = "\n\n".join(
        f"[{s['id']}] {s['title']}\nURL: {s['url']}\n{s['snippet']}" for s in sources
    )
    notes = _ask(
        "You are a research assistant. Using ONLY the search results given, write concise "
        "bullet-point notes about the topic. After each fact add its source number like [2]. "
        "Do not add facts that are not in the results.",
        f"Topic: {topic}\n\nSearch results:\n{results_text}",
    )
    return {
        "queries": queries,
        "sources": sources,
        "notes": notes,
        "log": state.get("log", []) + [f"Researcher: ran {len(queries)} searches, kept {len(sources)} sources."],
    }


def writer(state: ResearchState) -> dict:
    revision_block = ""
    if state.get("issues"):
        issues = "\n".join(f"- {i}" for i in state["issues"])
        revision_block = (
            f"\n\nYour previous draft:\n{state['draft']}\n\n"
            f"The Reviewer found these problems. Fix ALL of them (remove or correct the claims):\n{issues}"
        )

    draft = _ask(
        "You are an academic writer helping a college student. Write a clear summary using ONLY "
        "the notes provided. Structure: ## Overview, ## Key Points, ## Details, ## Conclusion. "
        "Keep the source citations like [1] next to the claims. If the notes are thin, say so "
        "instead of inventing information.",
        f"Topic: {state['topic']}\n\nNotes:\n{state['notes']}{revision_block}",
    )
    n = state.get("revisions", 0)
    label = "first draft" if not state.get("issues") else f"revision {n}"
    return {"draft": draft, "log": state.get("log", []) + [f"Writer: produced {label}."]}


def reviewer(state: ResearchState) -> dict:
    raw = _ask(
        "You are a strict fact-checker. Compare the DRAFT against the NOTES. A claim is "
        "unsupported if it does not appear in the notes. Respond with ONLY JSON like:\n"
        '{"verdict": "APPROVED" or "REVISE", "issues": ["issue 1", "issue 2"]}\n'
        "Use APPROVED only if every claim is supported by the notes.",
        f"NOTES:\n{state['notes']}\n\nDRAFT:\n{state['draft']}",
        temperature=0.0,
    )
    try:
        data = json.loads(re.search(r"\{.*\}", raw, re.S).group(0))
        verdict = "REVISE" if str(data.get("verdict", "")).upper() == "REVISE" else "APPROVED"
        issues = [str(i) for i in data.get("issues", [])]
    except Exception:
        verdict, issues = "APPROVED", ["Reviewer output could not be parsed; treated as approved."]

    revisions = state.get("revisions", 0) + (1 if verdict == "REVISE" else 0)
    return {
        "verdict": verdict,
        "issues": issues if verdict == "REVISE" else [],
        "revisions": revisions,
        "log": state.get("log", []) + [f"Reviewer: {verdict} ({len(issues)} issue(s))."],
    }


def route_after_review(state: ResearchState) -> str:
    if state["verdict"] == "REVISE" and state["revisions"] <= MAX_REVISIONS:
        return "writer"
    return END


# ------------------------------------------------------------------------ graph
def build_graph():
    g = StateGraph(ResearchState)
    g.add_node("researcher", researcher)
    g.add_node("writer", writer)
    g.add_node("reviewer", reviewer)
    g.set_entry_point("researcher")
    g.add_conditional_edges("researcher", lambda s: "writer" if s.get("notes") else END)
    g.add_edge("writer", "reviewer")
    g.add_conditional_edges("reviewer", route_after_review, {"writer": "writer", END: END})
    return g.compile()


def stream_research(topic: str):
    """Yields (node_name, state_update) as each agent finishes. Used by the UI."""
    app = build_graph()
    initial = {"topic": topic, "revisions": 0, "issues": [], "log": []}
    for step in app.stream(initial, stream_mode="updates"):
        for node, update in step.items():
            yield node, update


def run_research(topic: str) -> ResearchState:
    """Runs the full pipeline and returns the final merged state."""
    state: ResearchState = {"topic": topic, "revisions": 0, "issues": [], "log": []}
    for _, update in stream_research(topic):
        state.update(update)
    return state
```

### `app.py`

```python
import streamlit as st

from agents import MAX_REVISIONS, stream_research

st.set_page_config(page_title="Research Assistant", page_icon="📚", layout="wide")
st.title("📚 Research Assistant for College Students")
st.caption("Three agents: Researcher → Writer → Reviewer (fact-check loop)")

topic = st.text_input("Enter a research topic", placeholder="e.g. Impact of sleep on exam performance")
run = st.button("Research", type="primary", disabled=not topic.strip())

if run:
    state = {"revisions": 0, "issues": []}
    status_box = st.status("Agents are working...", expanded=True)

    labels = {
        "researcher": "🔎 Researcher finished: searched the web and wrote notes",
        "writer": "✍️ Writer finished: produced a draft",
        "reviewer": "🧐 Reviewer finished: checked the draft against the notes",
    }

    try:
        for node, update in stream_research(topic.strip()):
            state.update(update)
            msg = labels[node]
            if node == "reviewer":
                msg += f" → **{update['verdict']}**"
                for issue in update.get("issues", []):
                    msg += f"\n  - ⚠️ {issue}"
            status_box.write(msg)
        status_box.update(label="Done", state="complete", expanded=False)
    except Exception as e:
        status_box.update(label="Something went wrong", state="error")
        st.error(f"{type(e).__name__}: {e}\n\nCheck your GOOGLE_API_KEY in the .env file.")
        st.stop()

    if not state.get("notes"):
        st.warning("No search results were found for this topic. Try rephrasing it.")
        st.stop()

    if state.get("verdict") == "REVISE":
        st.warning(
            f"The Reviewer still had concerns after {MAX_REVISIONS} revisions. "
            "Treat the unsupported claims below with caution."
        )
        for issue in state.get("issues", []):
            st.write(f"- {issue}")
    else:
        st.success(f"Approved by Reviewer after {state['revisions']} revision(s).")

    tab1, tab2, tab3, tab4 = st.tabs(["Final Summary", "Researcher Notes", "Sources", "Agent Log"])
    with tab1:
        st.markdown(state["draft"])
        st.download_button("Download as Markdown", state["draft"], file_name="summary.md")
    with tab2:
        st.markdown(state["notes"])
        st.caption("Queries used: " + " | ".join(state.get("queries", [])))
    with tab3:
        for s in state.get("sources", []):
            st.markdown(f"**[{s['id']}]** [{s['title']}]({s['url']})")
    with tab4:
        for line in state.get("log", []):
            st.write("• " + line)
```

### `test_runner.py`

```python
"""
Runs the 5 planned test topics and saves evidence for the report.
Usage:  python test_runner.py
Output: test_outputs/<topic>.md for each topic + test_outputs/SUMMARY.md
"""
import re
import time
from pathlib import Path

from agents import run_research

TESTS = [
    ("Well-known topic", "Machine learning basics"),
    ("Recent topic", "Latest developments in solar energy"),
    ("Narrow topic", "Impact of sleep on exam performance"),
    ("Vague topic", "Technology"),
    ("Low-information topic", "History of the village of Mulshi's local festivals"),
]

out_dir = Path("test_outputs")
out_dir.mkdir(exist_ok=True)
rows = []

for category, topic in TESTS:
    print(f"\n=== {category}: {topic} ===")
    start = time.time()
    try:
        s = run_research(topic)
        secs = round(time.time() - start, 1)
        slug = re.sub(r"[^a-z0-9]+", "_", topic.lower()).strip("_")[:50]
        report = (
            f"# {topic}\n\n**Category:** {category}\n"
            f"**Verdict:** {s.get('verdict', 'N/A')}  |  **Revisions:** {s.get('revisions', 0)}  |  "
            f"**Sources:** {len(s.get('sources', []))}  |  **Time:** {secs}s\n\n"
            f"## Agent log\n" + "\n".join(f"- {l}" for l in s.get("log", []))
            + f"\n\n## Final draft\n\n{s.get('draft', '(no draft: no search results)')}\n"
        )
        (out_dir / f"{slug}.md").write_text(report, encoding="utf-8")
        rows.append((category, topic, s.get("verdict", "N/A"), s.get("revisions", 0),
                     len(s.get("sources", [])), secs))
        print("Verdict:", s.get("verdict"), "| Revisions:", s.get("revisions"))
    except Exception as e:
        rows.append((category, topic, f"ERROR: {type(e).__name__}", 0, 0, 0))
        print("Error:", e)

table = "| Category | Topic | Verdict | Revisions | Sources | Time (s) |\n|---|---|---|---|---|---|\n"
table += "\n".join("| " + " | ".join(str(c) for c in r) + " |" for r in rows)
(out_dir / "SUMMARY.md").write_text(table, encoding="utf-8")
print("\n" + table)
```
